/* single word (4) or double word (8) alignment */
#define ALIGNMENT 8

/////////////////////////////////////
/* 매크로 정의 */
#define WSIZE       4       /* 워드 및 헤더/푸터 크기 (바이트) */
#define DSIZE       8       /* 더블 워드 크기 (바이트)  */
#define CHUNKSIZE  (1<<12)  /* 힙 확장을 위한 기본 청크 크기 (4KB) */

#define MAX(x, y) ((x) > (y)? (x) : (y))

/* 헤더 또는 푸터에서 크기 및 할당 비트 읽기/쓰기 */
#define PACK(size, alloc)  ((size) | (alloc)) /* 크기와 할당 비트를 결합  */
#define GET(p)       (*(unsigned int *)(p)) /* 주소 p에서 워드 읽기 */
#define PUT(p, val)  (*(unsigned int *)(p) = (val)) /* 주소 p에 워드 쓰기 */

/* 주소 p에서 크기와 할당된 비트 읽기 */
#define GET_SIZE(p)  (GET(p) & ~0x7) /* 헤더/푸터에서 크기 추출 */
#define GET_ALLOC(p) (GET(p) & 0x1)  /* 헤더/푸터에서 할당 비트 추출 */

/* 블록 포인터 bp로부터 헤더 및 푸터 주소 계산 */
#define HDRP(bp)       ((char *)(bp) - WSIZE) /* 페이로드 포인터에서 헤더 포인터 계산 */
#define FTRP(bp)       ((char *)(bp) + GET_SIZE(HDRP(bp)) - DSIZE) /* 페이로드 포인터에서 푸터 포인터 계산 */

/* 블록 포인터 bp로부터 다음 및 이전 블록 포인터 계산 */
#define NEXT_BLKP(bp)  ((char *)(bp) + GET_SIZE(((char *)(bp) - WSIZE))) /* 현재 블록의 다음 블록 페이로드 포인터 */
#define PREV_BLKP(bp)  ((char *)(bp) - GET_SIZE(((char *)(bp) - DSIZE))) /* 현재 블록의 이전 블록 페이로드 포인터 */

/* 명시적 가용 리스트 관리를 위한 매크로 */
#define PRED_PTR(bp)   ((char *)(bp))         /* 가용 블록 bp에서 이전(previous) 포인터 저장 위치 */
#define SUCC_PTR(bp)   ((char *)(bp) + WSIZE) /* 가용 블록 bp에서 다음(successor) 포인터 저장 위치 */

/* 가용 블록의 이전/다음 블록 포인터 읽기 */
#define GET_PRED(bp)   (*(char **)(bp))         /* bp의 이전 가용 블록 주소 읽기 */
#define GET_SUCC(bp)   (*(char **)(SUCC_PTR(bp))) /* bp의 다음 가용 블록 주소 읽기 */

/* 가용 블록의 이전/다음 블록 포인터 쓰기 */
#define SET_PRED(bp, val)  (*(char **)(bp) = (char *)(val))
#define SET_SUCC(bp, val)  (*(char **)(SUCC_PTR(bp)) = (char *)(val))
/////////////////////////////////////////////
/* rounds up to the nearest multiple of ALIGNMENT */
#define ALIGN(size) (((size) + (ALIGNMENT-1)) & ~0x7)
#define OVERHEAD DSIZE /* 헤더와 푸터의 크기 (8바이트) */

#define SIZE_T_SIZE (ALIGN(sizeof(size_t)))



/* 전역 변수 */
static char *heap_listp; /* 힙의 시작을 가리키는 포인터 */
static char *free_listp; /* 명시적 가용 리스트의 헤드 포인터 (LIFO) */

/* 정적 함수 선언 */
static void *extend_heap(size_t words);
static void *coalesce(void *bp);
static void *find_fit(size_t asize);
static void place(void *bp, size_t asize);
static void add_to_free_list(void *bp);
static void remove_from_free_list(void *bp);



/* 
 * mm_init - initialize the malloc package.
 */
int mm_init(void)
{
    /* 빈 힙 생성: 패딩 + 프롤로그 헤더 + 프롤로그 푸터 + 에필로그 헤더 */
    if ((heap_listp = mem_sbrk(4*WSIZE)) == (void *)-1)
        return -1;

    PUT(heap_listp, 0);                             /* 정렬을 위한 패딩 */
    PUT(heap_listp + (1*WSIZE), PACK(DSIZE, 1));    /* 프롤로그 헤더 (크기=8바이트, 할당됨) */
    PUT(heap_listp + (2*WSIZE), PACK(DSIZE, 1));    /* 프롤로그 푸터 */
    PUT(heap_listp + (3*WSIZE), PACK(0, 1));        /* 에필로그 헤더 (크기=0, 할당됨) */
    //free_listp = heap_listp + (2*WSIZE); /* 초기 가용 리스트는 프롤로그 푸터부터 시작 (실제 가용 블록 없음) */

    // (수정할 코드)
    free_listp = NULL; /* 명시적 가용 리스트는 처음에는 비어있음 */
    /* CHUNKSIZE 바이트로 힙 확장 및 초기 가용 블록 생성 */
    if (extend_heap(CHUNKSIZE/WSIZE) == NULL)
        return -1;

    return 0;
}

/* 
 * mm_malloc - Allocate a block by incrementing the brk pointer.
 *     Always allocate a block whose size is a multiple of the alignment.
 */
void *mm_malloc(size_t size)
{
    size_t asize;      /* 조정된 블록 크기 */
    size_t extendsize; /* 힙에 맞지 않을 경우 확장할 크기 */
    char *bp;

    /* 가짜 요청 무시 */
    if (size == 0)
        return NULL;

    // /* 더블 워드 정렬 및 오버헤드(헤더/푸터)를 포함하도록 블록 크기 조정 */
    // if (size <= DSIZE) /* 최소 블록 크기: 16바이트 (헤더, 푸터 포함 8바이트 + 페이로드 8바이트) */
    //     asize = 2*DSIZE;
    // else /* (요청 크기 + 오버헤드)를 더블 워드 배수로 조정 */
    //     asize = DSIZE * ((size + (DSIZE) + (DSIZE-1)) / DSIZE);

    /* 가용 블록 크기가 요청 크기보다 작으면 확장 */
    
    if (size <= DSIZE) // 요청 크기가 8바이트 이하라면
        asize = 2 * DSIZE; // 최소 블록 크기 16바이트 (헤더+푸터 8 + 실제 페이로드 8)
    else // 요청 크기가 8바이트 초과라면
        asize = ALIGN(size + OVERHEAD); // 요청 크기 + 헤더/푸터 오버헤드 8바이트를 ALIGNMENT에 맞춤

    /* 가용 리스트에서 적합한 블록 검색 (First-Fit) */
    if ((bp = find_fit(asize)) != NULL) {
        place(bp, asize); /* 찾은 블록 할당 및 필요시 분할 */
        return bp;
    }

    /* 적합한 블록을 찾지 못함. 힙 확장 후 새로운 블록 할당 */
    extendsize = MAX(asize, CHUNKSIZE);
    if ((bp = extend_heap(extendsize/WSIZE)) == NULL)
        return NULL;
    place(bp, asize); /* 확장된 힙에서 블록 할당 */

    return bp;
}

/*
 * mm_free - Freeing a block does nothing.
 */
void mm_free(void *ptr)
{
    size_t size = GET_SIZE(HDRP(ptr)); /* 블록의 크기 가져오기 */

    PUT(HDRP(ptr), PACK(size, 0)); /* 헤더를 가용으로 표시 */
    PUT(FTRP(ptr), PACK(size, 0)); /* 푸터를 가용으로 표시 */
    coalesce(ptr); /* 인접한 가용 블록과 병합 */

}

/*
 * mm_realloc - Implemented simply in terms of mm_malloc and mm_free
 */
void *mm_realloc(void *ptr, size_t size)
{
 void *newptr;
    size_t oldsize;

    /* 1. ptr이 NULL이면 mm_malloc(size)와 동일합니다.  */
    if (ptr == NULL) {

        return mm_malloc(size);
    }

    /* 2. size가 0이면 mm_free(ptr)와 동일합니다.  */
    if (size == 0) {
        mm_free(ptr);

        return NULL;
    }

    oldsize = GET_SIZE(HDRP(ptr)); /* 기존 블록의 전체 크기 */
    size_t new_asize = DSIZE * ((size + (DSIZE) + (DSIZE-1)) / DSIZE); /* 요청 size에 대한 새로운 할당 크기 */

    /* 새 크기가 기존 블록의 크기와 동일하거나 더 작다면 */
    if (new_asize <= oldsize) {
        // 이 부분에서 블록을 축소하고 남은 부분을 가용 블록으로 만들 수 있지만,

        // 더 정교한 구현을 위해서는 여기에서 분할 로직이 들어갈 수 있습니다.
        return ptr;
    } else { /* 새 크기가 기존 블록보다 더 크다면 */
        size_t next_alloc = GET_ALLOC(HDRP(NEXT_BLKP(ptr))); /* 다음 블록의 할당 여부 */
        size_t prev_alloc = GET_ALLOC(FTRP(PREV_BLKP(ptr))); // 이전 블록의 할당 여부


        size_t current_and_next_size = oldsize + GET_SIZE(HDRP(NEXT_BLKP(ptr))); /* 현재 블록과 다음 블록을 합친 크기 */

        /* 다음 블록이 가용 상태이고, 두 블록을 합치면 요청 크기를 만족한다면 */
        if (!next_alloc && (current_and_next_size >= new_asize)) {
            remove_from_free_list(NEXT_BLKP(ptr)); /* 다음 가용 블록을 가용 리스트에서 제거 */
            PUT(HDRP(ptr), PACK(current_and_next_size, 1)); /* 현재 블록의 헤더 업데이트 */
            PUT(FTRP(ptr), PACK(current_and_next_size, 1)); /* 새 푸터 업데이트 */
            return ptr; /* 기존 포인터 반환 */
        } 
            newptr = mm_malloc(size); /* 새로운 블록 할당 */
            if (newptr == NULL) {

                return NULL;
            }
            //memcpy(newptr, ptr, size); /* 기존 내용 복사  */
            // size_t copy_size = (size < (oldsize - DSIZE)) ? size : (oldsize - DSIZE); // 요청 크기 또는 기존 페이로드 크기 중 작은 값
            size_t old_payload_size = GET_SIZE(HDRP(ptr)) - DSIZE;
            // 복사할 크기는 요청된 새 크기와 원래 블록의 실제 페이로드 크기 중 작은 값
            size_t copy_size = (size < old_payload_size) ? size : old_payload_size;
            memcpy(newptr, ptr, copy_size);
            mm_free(ptr); /* 기존 블록 해제 */

            return newptr;
        }
    
}


/*
 * extend_heap - 워드 단위로 힙을 확장합니다.
 * 워드 수만큼 힙을 확장하고, 새로운 가용 블록을 생성한 뒤 병합을 시도합니다.
 */
static void *extend_heap(size_t words)
{
    char *bp;
    size_t size;

    /* 더블 워드 정렬을 유지하며 힙 확장 */
    size = (words % 2) ? (words + 1) * WSIZE : words * WSIZE;
    if ((long)(bp = mem_sbrk(size)) == -1)
        return NULL;

    /* 새로운 가용 블록 헤더/푸터 설정 및 에필로그 헤더 이동 */
    PUT(HDRP(bp), PACK(size, 0));         /* 가용 블록 헤더 */
    PUT(FTRP(bp), PACK(size, 0));         /* 가용 블록 푸터 */
    PUT(HDRP(NEXT_BLKP(bp)), PACK(0, 1)); /* 새로운 에필로그 헤더 */

    /* 이전 블록이 가용 상태였다면 병합 */
    return coalesce(bp);
}

/*
 * coalesce - 인접한 가용 블록과 병합하여 단편화를 줄입니다.
 */
static void *coalesce(void *bp)
{
    assert(bp != NULL);
    assert((char *)bp >= (char *)mem_heap_lo());
    assert((char *)bp < (char *)mem_heap_hi());

    // 최소 블록 크기보다 작은 주소면 무조건 이상한 값
    if ((char *)bp < heap_listp + 2 * WSIZE) {
        fprintf(stderr, "[coalesce] ERROR: bp (%p) is before first real block.\n", bp);
        return NULL;
    }

    size_t prev_alloc;
    size_t next_alloc = GET_ALLOC(HDRP(NEXT_BLKP(bp)));
    size_t size = GET_SIZE(HDRP(bp));

    //printf("[coalesce] bp: %p, HDRP(bp): %p\n", bp, HDRP(bp));
    //printf("[coalesce] heap_listp: %p, mem_heap_lo(): %p\n", heap_listp, mem_heap_lo());

    // char *first_payload = heap_listp + 2 * WSIZE; // 프롤로그 푸터 다음이 첫 페이로드
    // if ((char *)bp <= first_payload) {
    //     //printf("[coalesce] First block or invalid PREV_BLKP — skipping backward check\n");
    //     prev_alloc = 1;
    // } else {
    //     char *prev_bp = PREV_BLKP(bp);
    //     //printf("[coalesce] PREV_BLKP(bp): %p\n", prev_bp);
    //     assert(prev_bp >= (char *)mem_heap_lo());
    //     assert(prev_bp < (char *)mem_heap_hi());
    //     prev_alloc = GET_ALLOC(FTRP(prev_bp));
    // }

    char *first_payload = heap_listp + 2 * WSIZE; // 프롤로그 푸터 다음이 첫 페이로드
    if ((char *)bp <= first_payload) {
        prev_alloc = 1; // 프롤로그 블록이거나 그 이전이면 항상 할당된 것으로 간주
    } else {
        char *prev_bp = PREV_BLKP(bp);
        prev_alloc = GET_ALLOC(FTRP(prev_bp));
    }

    // 다음 블록 할당 상태 디버깅 로그
    //printf("[coalesce] next_alloc: %zu, size: %zu\n", next_alloc, size);


    /* 4가지 병합 케이스 */
    if (prev_alloc && next_alloc) {         /* Case 1: 이전/다음 블록 모두 할당됨 */
        add_to_free_list(bp); /* 현재 블록만 가용 리스트에 추가 */
        return bp;
    }

    else if (prev_alloc && !next_alloc) { /* Case 2: 이전 블록 할당, 다음 블록 가용 */
        remove_from_free_list(NEXT_BLKP(bp)); /* 다음 가용 블록을 리스트에서 제거 */
        size += GET_SIZE(HDRP(NEXT_BLKP(bp))); /* 현재 블록 크기에 다음 블록 크기 추가 */
        PUT(HDRP(bp), PACK(size, 0)); /* 현재 블록 헤더 업데이트 */
        PUT(FTRP(bp), PACK(size, 0)); /* 다음 블록 푸터에 새로운 푸터 쓰기 */
        add_to_free_list(bp); /* 병합된 현재 블록을 가용 리스트에 추가 */
    }

    else if (!prev_alloc && next_alloc) { /* Case 3: 이전 블록 가용, 다음 블록 할당 */
        remove_from_free_list(PREV_BLKP(bp)); /* 이전 가용 블록을 리스트에서 제거 */
        size += GET_SIZE(HDRP(PREV_BLKP(bp))); /* 현재 블록 크기에 이전 블록 크기 추가 */
        PUT(FTRP(bp), PACK(size, 0)); /* 현재 블록 푸터 업데이트 */
        PUT(HDRP(PREV_BLKP(bp)), PACK(size, 0)); /* 이전 블록 헤더에 새로운 헤더 쓰기 */
        bp = PREV_BLKP(bp); /* 블록 포인터를 이전 블록으로 이동 */
        add_to_free_list(bp); /* 병합된 이전 블록을 가용 리스트에 추가 */
    }

    else {                                  /* Case 4: 이전/다음 블록 모두 가용 */
        remove_from_free_list(PREV_BLKP(bp)); /* 이전 가용 블록을 리스트에서 제거 */
        remove_from_free_list(NEXT_BLKP(bp)); /* 다음 가용 블록을 리스트에서 제거 */
        size += GET_SIZE(HDRP(PREV_BLKP(bp))) + GET_SIZE(FTRP(NEXT_BLKP(bp))); /* 세 블록의 크기 합산 */
        PUT(HDRP(PREV_BLKP(bp)), PACK(size, 0)); /* 이전 블록 헤더에 새로운 헤더 쓰기 */
        PUT(FTRP(NEXT_BLKP(bp)), PACK(size, 0)); /* 다음 블록 푸터에 새로운 푸터 쓰기 */
        bp = PREV_BLKP(bp); /* 블록 포인터를 이전 블록으로 이동 */
        add_to_free_list(bp); /* 병합된 블록을 가용 리스트에 추가 */
    }
    return bp;
}

/*
 * find_fit - First-Fit 방식으로 가용 리스트에서 asize를 수용할 수 있는 블록을 찾습니다.
 */
static void *find_fit(size_t asize)
{
    /* 명시적 가용 리스트를 순회하며 적합한 블록 검색 */
    void *bp;
    for (bp = free_listp; bp != NULL; bp = GET_SUCC(bp)) {
        if (asize <= GET_SIZE(HDRP(bp))) {
            return bp; // 적합한 블록을 찾으면 반환
        }
    }
    return NULL; /* 적합한 블록을 찾지 못함 */
}

/*
 * place - 찾은 가용 블록 bp에 asize 크기의 할당된 블록을 배치하고,
 * 남은 공간이 있다면 새로운 가용 블록으로 분할합니다.
 */
static void place(void *bp, size_t asize)
{
    size_t csize = GET_SIZE(HDRP(bp)); /* 현재 가용 블록의 크기 */

    remove_from_free_list(bp); /* 가용 리스트에서 현재 블록 제거 */

    if ((csize - asize) >= (2*DSIZE)) { /* 블록 분할이 가능한 경우 (최소 블록 크기 16바이트) */
        PUT(HDRP(bp), PACK(asize, 1));     /* 할당된 블록 헤더 설정 */
        PUT(FTRP(bp), PACK(asize, 1));     /* 할당된 블록 푸터 설정 */
        bp = NEXT_BLKP(bp);                 /* 다음 블록으로 이동 */
        PUT(HDRP(bp), PACK(csize - asize, 0)); /* 남은 공간에 대한 가용 블록 헤더 설정 */
        PUT(FTRP(bp), PACK(csize - asize, 0)); /* 남은 공간에 대한 가용 블록 푸터 설정 */
        coalesce(bp); /* 분할된 새 가용 블록 병합 시도 (이전 블록은 할당되었으므로 Case 1 또는 2) */
    } else { /* 분할할 수 없는 경우 (내부 단편화 발생) */
        PUT(HDRP(bp), PACK(csize, 1)); /* 전체 블록을 할당된 것으로 표시 */
        PUT(FTRP(bp), PACK(csize, 1));
    }
}

/*
 * add_to_free_list - LIFO 방식으로 가용 블록을 가용 리스트 헤드에 추가합니다.
 */
static void add_to_free_list(void *bp)
{
    SET_SUCC(bp, free_listp); // 새 블록의 다음 포인터는 현재 헤드
    if (free_listp != NULL) { // 리스트가 비어있지 않았다면, 이전 헤드의 이전 포인터 업데이트
        SET_PRED(free_listp, bp);
    }
    SET_PRED(bp, NULL);       // 새 블록은 새로운 헤드이므로 이전 포인터는 NULL
    free_listp = bp;          // 헤드 포인터 업데이트
}

/*
 * remove_from_free_list - 가용 블록을 가용 리스트에서 제거합니다.
 */
static void remove_from_free_list(void *bp)
{
    if (GET_PRED(bp) == NULL) { /* 리스트의 헤드인 경우 */
        free_listp = GET_SUCC(bp);
        if (GET_SUCC(bp) != NULL) { /* 다음 블록이 있다면 */
            SET_PRED(GET_SUCC(bp), NULL);
        }
    } else { /* 리스트의 중간 또는 끝에 있는 경우 */
        SET_SUCC(GET_PRED(bp), GET_SUCC(bp)); /* 이전 블록의 다음 포인터 업데이트 */
        if (GET_SUCC(bp) != NULL) { /* 다음 블록이 있다면 */
            SET_PRED(GET_SUCC(bp), GET_PRED(bp)); /* 다음 블록의 이전 포인터 업데이트 */
        }
    }
}
